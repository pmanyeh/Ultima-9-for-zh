cmake_minimum_required(VERSION 3.24)
project(U9ChtRuntime VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "U9Cht requires MSVC Win32/x86: configure with -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(U9ChtCommon STATIC common/platform.cpp)
target_include_directories(U9ChtCommon PUBLIC common)
target_link_libraries(U9ChtCommon PUBLIC bcrypt)
add_executable(U9ChtLauncher U9ChtLauncher/main.cpp)
target_link_libraries(U9ChtLauncher PRIVATE U9ChtCommon)
add_library(U9ChtRuntime SHARED U9ChtRuntime/runtime.cpp U9ChtRuntime/exports.def)
add_library(U9ChtHooks STATIC U9ChtRuntime/hook_manager.cpp)
target_include_directories(U9ChtHooks PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtHooks PUBLIC U9ChtCommon)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtHooks)
add_library(U9ChtIdentity STATIC U9ChtRuntime/identity.cpp)
target_include_directories(U9ChtIdentity PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtIdentity PUBLIC U9ChtCommon)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtIdentity)
add_library(U9ChtPixels STATIC U9ChtRuntime/compositor.cpp U9ChtRuntime/glyph_cache.cpp)
target_include_directories(U9ChtPixels PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtPixels PUBLIC U9ChtCommon gdi32)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtPixels)
add_library(U9ChtEngine STATIC U9ChtRuntime/engine.cpp)
target_include_directories(U9ChtEngine PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtEngine PUBLIC U9ChtHooks U9ChtIdentity U9ChtPixels psapi)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtEngine)
enable_testing()
add_executable(U9ChtNativeTests tests/native_tests.cpp)
target_link_libraries(U9ChtNativeTests PRIVATE U9ChtCommon)
add_test(NAME U9ChtNativeTests COMMAND U9ChtNativeTests)
add_executable(U9ChtHookTests tests/hook_tests.cpp)
target_link_libraries(U9ChtHookTests PRIVATE U9ChtHooks)
add_test(NAME U9ChtHookTests COMMAND U9ChtHookTests)
add_executable(U9ChtIdentityTests tests/identity_tests.cpp)
target_link_libraries(U9ChtIdentityTests PRIVATE U9ChtIdentity)
add_test(NAME U9ChtIdentityTests COMMAND U9ChtIdentityTests)
add_executable(U9ChtPixelTests tests/pixel_tests.cpp)
target_link_libraries(U9ChtPixelTests PRIVATE U9ChtPixels)
add_test(NAME U9ChtPixelTests COMMAND U9ChtPixelTests)
add_executable(U9ChtShutdownTests tests/shutdown_tests.cpp)
target_link_libraries(U9ChtShutdownTests PRIVATE U9ChtEngine)
add_test(NAME U9ChtShutdownTests COMMAND U9ChtShutdownTests)

cmake_minimum_required(VERSION 3.24)
project(U9ChtRuntime VERSION 0.2.40 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "U9Cht requires MSVC Win32/x86: configure with -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(U9ChtCommon STATIC common/platform.cpp)
target_include_directories(U9ChtCommon PUBLIC common)
target_link_libraries(U9ChtCommon PUBLIC bcrypt)
add_executable(U9ChtLauncher U9ChtLauncher/main.cpp)
target_link_libraries(U9ChtLauncher PRIVATE U9ChtCommon)
add_library(U9ChtRuntime SHARED U9ChtRuntime/runtime.cpp U9ChtRuntime/exports.def)
add_library(U9ChtHooks STATIC U9ChtRuntime/hook_manager.cpp)
target_include_directories(U9ChtHooks PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtHooks PUBLIC U9ChtCommon)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtHooks)
add_library(U9ChtIdentity STATIC U9ChtRuntime/identity.cpp)
target_include_directories(U9ChtIdentity PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtIdentity PUBLIC U9ChtCommon)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtIdentity)
add_library(U9ChtPixels STATIC U9ChtRuntime/compositor.cpp U9ChtRuntime/glyph_cache.cpp)
target_include_directories(U9ChtPixels PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtPixels PUBLIC U9ChtCommon gdi32)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtPixels)
add_library(U9ChtEngine STATIC U9ChtRuntime/engine.cpp U9ChtRuntime/input_hooks.cpp)
target_include_directories(U9ChtEngine PUBLIC U9ChtRuntime)
target_link_libraries(U9ChtEngine PUBLIC U9ChtHooks U9ChtIdentity U9ChtPixels U9ChtMovie strmiids psapi)
target_link_libraries(U9ChtRuntime PRIVATE U9ChtEngine)
# P11 movie player: pl_mpeg (MIT, third_party) + GDI + waveOut. Shared by the runtime and the review tool.
add_library(U9ChtMovie STATIC U9ChtMovie/movie_player.cpp U9ChtMovie/movie_graph.cpp U9ChtMovie/pl_mpeg_impl.c)
set_source_files_properties(U9ChtMovie/pl_mpeg_impl.c PROPERTIES COMPILE_OPTIONS "/W0;/WX-")
target_include_directories(U9ChtMovie PUBLIC U9ChtMovie third_party/pl_mpeg)
target_link_libraries(U9ChtMovie PUBLIC winmm gdi32 user32 msimg32)
add_executable(U9ChtMovieViewer WIN32 U9ChtMovieViewer/main.cpp)
target_link_libraries(U9ChtMovieViewer PRIVATE U9ChtMovie U9ChtPixels shell32)
add_executable(U9ChtMovieProbe tools/movie_probe.cpp)
target_link_libraries(U9ChtMovieProbe PRIVATE U9ChtMovie)
enable_testing()
add_executable(U9ChtMovieSubtitleTests tests/movie_subtitle_tests.cpp)
target_include_directories(U9ChtMovieSubtitleTests PRIVATE U9ChtMovie)
add_test(NAME U9ChtMovieSubtitleTests COMMAND U9ChtMovieSubtitleTests)
add_executable(U9ChtNativeTests tests/native_tests.cpp)
target_link_libraries(U9ChtNativeTests PRIVATE U9ChtCommon)
add_test(NAME U9ChtNativeTests COMMAND U9ChtNativeTests)
add_executable(U9ChtHookTests tests/hook_tests.cpp)
target_link_libraries(U9ChtHookTests PRIVATE U9ChtHooks)
add_test(NAME U9ChtHookTests COMMAND U9ChtHookTests)
add_executable(U9ChtIdentityTests tests/identity_tests.cpp)
target_link_libraries(U9ChtIdentityTests PRIVATE U9ChtIdentity)
add_test(NAME U9ChtIdentityTests COMMAND U9ChtIdentityTests)
add_executable(U9ChtRasterRunTests tests/raster_run_tests.cpp)
target_link_libraries(U9ChtRasterRunTests PRIVATE U9ChtIdentity)
add_test(NAME U9ChtRasterRunTests COMMAND U9ChtRasterRunTests)
add_executable(U9ChtMiscMarkerTests tests/misc_marker_tests.cpp)
target_include_directories(U9ChtMiscMarkerTests PRIVATE U9ChtRuntime)
add_test(NAME U9ChtMiscMarkerTests COMMAND U9ChtMiscMarkerTests)
add_executable(U9ChtClockTextTests tests/clock_text_tests.cpp)
target_include_directories(U9ChtClockTextTests PRIVATE U9ChtRuntime)
add_test(NAME U9ChtClockTextTests COMMAND U9ChtClockTextTests)
add_executable(U9ChtPixelTests tests/pixel_tests.cpp)
target_link_libraries(U9ChtPixelTests PRIVATE U9ChtPixels)
add_test(NAME U9ChtPixelTests COMMAND U9ChtPixelTests)
add_executable(U9ChtShutdownTests tests/shutdown_tests.cpp)
target_link_libraries(U9ChtShutdownTests PRIVATE U9ChtEngine)
add_test(NAME U9ChtShutdownTests COMMAND U9ChtShutdownTests)
add_executable(U9ChtShowcaseCheck tools/showcase_check.cpp)
target_link_libraries(U9ChtShowcaseCheck PRIVATE U9ChtIdentity U9ChtPixels)
